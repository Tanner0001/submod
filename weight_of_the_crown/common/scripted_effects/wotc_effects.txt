wotc_init_variables = {
	if = {
		limit = { check_variable = { wotc_exile_influence = 0 } NOT = { has_country_flag = wotc_vars_initialized } }
		set_variable = { wotc_exile_influence = 50 } # start at the King's Compromise
	}
	set_variable = { wotc_border_tension = 0 }
	set_country_flag = wotc_vars_initialized
}

# Call after any add_to_variable = { wotc_exile_influence = X } to (re)apply the
# correct tier's ideas with hysteresis, so the tier only flips once the value
# has crossed a threshold by a margin, not right at the boundary.
wotc_reapply_exile_tier = {
	if = {
		limit = {
			check_variable = { wotc_exile_influence < 27 }
			NOT = { has_idea = wotc_idea_sovereignty }
		}
		wotc_clear_exile_tier_ideas = yes
		add_ideas = wotc_idea_sovereignty
	}
	else_if = {
		limit = {
			check_variable = { wotc_exile_influence > 73 }
			NOT = { has_idea = wotc_idea_exile_dominance }
		}
		wotc_clear_exile_tier_ideas = yes
		add_ideas = wotc_idea_exile_dominance
	}
	else_if = {
		limit = {
			check_variable = { wotc_exile_influence >= 27 }
			check_variable = { wotc_exile_influence <= 73 }
			NOT = { has_idea = wotc_idea_kings_compromise }
		}
		wotc_clear_exile_tier_ideas = yes
		add_ideas = wotc_idea_kings_compromise
	}
}

wotc_clear_exile_tier_ideas = {
	if = { limit = { has_idea = wotc_idea_sovereignty } remove_ideas = wotc_idea_sovereignty }
	if = { limit = { has_idea = wotc_idea_kings_compromise } remove_ideas = wotc_idea_kings_compromise }
	if = { limit = { has_idea = wotc_idea_exile_dominance } remove_ideas = wotc_idea_exile_dominance }
}

# Same hysteresis pattern for the 49th Parallel tension tiers.
wotc_reapply_tension_tier = {
	if = {
		limit = {
			check_variable = { wotc_border_tension < 31 }
			NOT = { has_idea = wotc_idea_tension_observation }
		}
		wotc_clear_tension_tier_ideas = yes
		add_ideas = wotc_idea_tension_observation
	}
	else_if = {
		limit = {
			check_variable = { wotc_border_tension > 69 }
			NOT = { has_idea = wotc_idea_tension_intervention }
		}
		wotc_clear_tension_tier_ideas = yes
		add_ideas = wotc_idea_tension_intervention
	}
	else_if = {
		limit = {
			check_variable = { wotc_border_tension >= 31 }
			check_variable = { wotc_border_tension <= 69 }
			NOT = { has_idea = wotc_idea_tension_covert_proxy }
		}
		wotc_clear_tension_tier_ideas = yes
		add_ideas = wotc_idea_tension_covert_proxy
	}
}

wotc_clear_tension_tier_ideas = {
	if = { limit = { has_idea = wotc_idea_tension_observation } remove_ideas = wotc_idea_tension_observation }
	if = { limit = { has_idea = wotc_idea_tension_covert_proxy } remove_ideas = wotc_idea_tension_covert_proxy }
	if = { limit = { has_idea = wotc_idea_tension_intervention } remove_ideas = wotc_idea_tension_intervention }
}

# Slow passive decay so a temporary border incident doesn't lock in permanent
# high alert. Called from events/wotc_canada.txt's self-rescheduling pulse
# event (wotc_canada.1), the standard Paradox-modding idiom for "run this
# periodically" since there's no generic on_action for it.
wotc_decay_border_tension = {
	if = {
		limit = { check_variable = { wotc_border_tension > 0 } }
		subtract_from_variable = { wotc_border_tension = 1 }
	}
}

### --- 49th Parallel: RCMP Emergency Response Team template --- ###
# A genuinely new division template, not a reuse of "District Militia".
# Pattern copied directly from the base game's own real precedent -
# common/scripted_effects/00_initialize_templates_from_techs.txt's
# `initialize_ranger_template` (fired by rangers_tech's on_research_complete)
# defines a brand-new template purely via a `division_template = { ... }`
# effect block, no history-file entry required. This does the same thing
# under our own name instead of reusing the vanilla "Rangers" template.
# ranger_battalion (common/units/infantry.txt) is real, special-forces-
# flagged, and thematically correct for an RCMP tactical unit - it's gated
# behind the "No Compromise, No Surrender" DLC (has_dlc_ncns, confirmed via
# common/doctrines/subdoctrines/land/infantry_subdoctrines.txt) and the
# rangers_tech tech (common/technologies/infantry.txt), which the decision
# force-researches via set_technology so it's available immediately without
# the player needing to have researched it manually.
wotc_initialize_rcmp_ert_template = {
	hidden_effect = {
		division_template = {
			name = "RCMP Emergency Response Team"
			regiments = {
				ranger_battalion = { x = 0 y = 0 }
				ranger_battalion = { x = 0 y = 1 }
				ranger_battalion = { x = 0 y = 2 }
			}
			priority = 2
		}
	}
}
