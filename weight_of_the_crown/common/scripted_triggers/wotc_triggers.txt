### Weight of the Crown - shared triggers ###
### Scope: country, unless noted otherwise ###

# True while the country is playing as the exiled Crown seated in Canada
# (CAN is a tag alias of IMP, active only while IMP_is_canada is set —
# see common/country_tag_aliases/tag_aliases.txt)
wotc_is_canada = {
	tag = CAN
}

# --- Exiles vs. Autonomy tiers (wotc_exile_influence, 0-100) ---
wotc_tier_sovereignty = {
	wotc_exile_influence < 30
}

wotc_tier_compromise = {
	wotc_exile_influence >= 30
	wotc_exile_influence <= 69
}

wotc_tier_exile_dominance = {
	wotc_exile_influence > 69
}

# --- 49th Parallel border-state check (state scope) ---
# Confirmed against this repo's own localisation/history files: Washington
# 386, Idaho 387, Montana 388, North Dakota 389, Minnesota 391, Michigan
# 393, Ohio 261, Maine 30/357, Vermont 933, New Hampshire 934.
#
# New York specifically checked and NOT resolvable from this repo: the only
# New-York-named state here is 358 "New York City" (history/states/358 -
# New York City.txt, owned by USA, 7.75M manpower) - that's the NYC metro
# state carved out by KR, not the upstate region that actually borders
# Canada. The upstate border state must still be a vanilla (unrenamed)
# state this repo doesn't ship its own copy of, so its id can't be
# confirmed here - needs the base game's state files or an in-game check,
# not a guess.
wotc_is_49th_parallel_state = {
	OR = {
		id = 386 # Washington
		id = 387 # Idaho
		id = 388 # Montana
		id = 389 # North Dakota
		id = 391 # Minnesota
		id = 393 # Michigan
		id = 261 # Ohio
		id = 30  # Maine
		id = 357 # Maine
		id = 933 # Vermont
		id = 934 # New Hampshire
		# id = XXX # upstate New York - still unresolved, see note above
	}
}

# True (state scope) if the state's current controller is hostile to Canada
# for 49th Parallel purposes. Confirmed hostile factions: WCA (Workers
# Congress of America, syndicalist) and APG (American Peoples Government,
# national_populist). USA and ACC are not treated as hostile.
wotc_state_controller_is_hostile_to_canada = {
	OR = {
		controller = { has_war_with = CAN }
		controller = { original_tag = WCA }
		controller = { original_tag = APG }
	}
}

# --- Border tension tiers (wotc_border_tension, 0-100) ---
wotc_tier_observation = {
	wotc_border_tension < 34
}

wotc_tier_covert_proxy = {
	wotc_border_tension >= 34
	wotc_border_tension <= 66
}

wotc_tier_intervention = {
	wotc_border_tension > 66
}
