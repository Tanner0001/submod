### Weight of the Crown - shared triggers ###
### Scope: country, unless noted otherwise ###

# True while the country is playing as the exiled Crown seated in Canada
# (CAN is a tag alias of IMP, active only while IMP_is_canada is set —
# see common/country_tag_aliases/tag_aliases.txt)
wotc_is_canada = {
	tag = CAN
}

# --- Exiles vs. Autonomy tiers (wotc_exile_influence, 0-100) ---
wotc_tier_sovereignty = {
	wotc_exile_influence < 30
}

wotc_tier_compromise = {
	wotc_exile_influence >= 30
	wotc_exile_influence <= 69
}

wotc_tier_exile_dominance = {
	wotc_exile_influence > 69
}

# True (state scope) if the state's current controller is hostile to Canada
# for 49th Parallel purposes. This repo's 2ACW belligerents are USA, ACC
# (American Constitutional Coalition), WCA, and APG (American Peoples
# Government) - there is no "CSA" tag in this version of Kaiserreich. Fill in
# whichever of these (or their dynamic-tag equivalents) should count as
# hostile to Canada before relying on this trigger; direct has_war_with is
# the only part of this that's solid without that confirmation.

# --- Border tension tiers (wotc_border_tension, 0-100) ---
wotc_tier_observation = {
	wotc_border_tension < 34
}

wotc_tier_covert_proxy = {
	wotc_border_tension >= 34
	wotc_border_tension <= 66
}

wotc_tier_intervention = {
	wotc_border_tension > 66
}
