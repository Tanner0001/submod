### Weight of the Crown - 49th Parallel Escalation ###

wotc_49th_parallel = {
	# One-time only (not repeatable - see wotc_rcmp_ert_deployed flag), or
	# this would spawn 3 free divisions every 45 days. Spawns real RCMP
	# Emergency Response Team garrisons at three border-facing cities,
	# using Canada's own existing "District Militia" division template
	# (history/units/CAN.txt - a real, already-defined small garrison
	# template, not invented here) at three verified real locations:
	# Toronto (state 276), Quebec/Montreal (state 832), Winnipeg/Manitoba
	# (state 467) - all confirmed via this repo's own victory-point
	# localisation and history/states files, not guessed.
	wotc_border_guard_deployment = {
		icon = GFX_decision_generic_defence
		available = {
			wotc_tier_observation = yes
			NOT = { has_country_flag = wotc_rcmp_ert_deployed }
		}
		cost = 25

		complete_effect = {
			# Does NOT grant CAN_refugee_crisis_help - confirmed the base
			# mod already grants that through its own event chain
			# (events/CAN events (Canada).txt), on the actual refugee
			# crisis mechanic. Duplicating it here would double-grant the
			# same idea outside that mechanic's real conditions.
			add_political_power = -25
			set_country_flag = wotc_rcmp_ert_deployed

			276 = { # Toronto
				create_unit = {
					division = "name = \"RCMP Emergency Response Team\" division_template = \"District Militia\" start_experience_factor = 0.8"
					owner = ROOT
				}
			}
			832 = { # Quebec (Montreal / Quebec City)
				create_unit = {
					division = "name = \"RCMP Emergency Response Team\" division_template = \"District Militia\" start_experience_factor = 0.8"
					owner = ROOT
				}
			}
			467 = { # Winnipeg (Manitoba)
				create_unit = {
					division = "name = \"RCMP Emergency Response Team\" division_template = \"District Militia\" start_experience_factor = 0.8"
					owner = ROOT
				}
			}
		}

		ai_will_do = { factor = 1 }
	}

	wotc_covert_infantry_aid = {
		icon = GFX_decision_generic_send_supplies
		available = {
			wotc_tier_covert_proxy = yes
		}
		cost = 40
		days_re_enable = 30

		complete_effect = {
			add_equipment_to_stockpile = {
				type = infantry_equipment_1
				amount = -200
				producer = CAN
			}
			add_to_variable = { wotc_border_tension = 3 } # covert action carries its own escalation risk
		}

		ai_will_do = { factor = 1 }
	}

	# NOTE: HOI4's Operations system (common/operations/) is occupation- and
	# resistance-network based (see operation_targeted_sabotage_infrastructure)
	# - it can't be launched against a state Canada doesn't occupy, which rules
	# it out for sabotaging a live hostile country's rail hubs from a decision.
	# This instead applies a dynamic state modifier (wotc_49th_parallel_sabotage_modifier,
	# built from real keys used elsewhere in this repo) to every currently
	# hostile-controlled border state, which is the closest verified
	# equivalent to the GDD's intent.
	wotc_sabotage_rail_hubs = {
		icon = GFX_decision_generic_sabotage
		available = {
			wotc_tier_covert_proxy = yes
		}
		cost = 60
		days_re_enable = 60

		complete_effect = {
			every_state = {
				limit = {
					wotc_is_49th_parallel_state = yes
					wotc_state_controller_is_hostile_to_canada = yes
				}
				set_state_flag = { flag = wotc_sabotage_active days = 60 }
				add_dynamic_modifier = { modifier = wotc_49th_parallel_sabotage_modifier }
			}
			add_to_variable = { wotc_border_tension = 3 } # covert action carries its own escalation risk
		}

		ai_will_do = { factor = 0.5 }
	}

	# NOTE: `create_wargoal` issued directly from a decision effect (as seen
	# in this repo's "01 China decisions.txt", e.g. china_attack_decision's
	# remove_effect) never goes through the World Tension-gated declare-war
	# diplomatic action at all - it's the standard way KR content already
	# bypasses WT, not a special trick. No separate "bypass" effect exists to
	# copy because this already is that pattern.
	wotc_intervention_war_justification = {
		icon = GFX_decision_generic_war
		available = {
			wotc_tier_intervention = yes
			OR = {
				AND = { country_exists = WCA NOT = { has_war_with = WCA } }
				AND = { country_exists = APG NOT = { has_war_with = APG } }
			}
		}
		cost = 150

		complete_effect = {
			# type = topple_government instead of annex_everything - confirmed
			# real (used by many KR ideological-intervention focuses, e.g.
			# common/national_focus/POL focus (Poland).txt,
			# RUS focus (Russia).txt) and fits "abandon restraint and prepare
			# direct intervention" much better than annexing US territory
			# outright would.
			if = {
				limit = { country_exists = WCA NOT = { has_war_with = WCA } }
				create_wargoal = { target = WCA type = topple_government }
			}
			if = {
				limit = { country_exists = APG NOT = { has_war_with = APG } }
				create_wargoal = { target = APG type = topple_government }
			}
			if = {
				limit = { NOT = { has_government = democratic } }
				add_stability = -0.15
			}
			else = {
				add_stability = -0.05
			}
		}

		ai_will_do = { factor = 0.2 }
	}
}
