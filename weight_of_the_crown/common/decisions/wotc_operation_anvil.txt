### Weight of the Crown - Operation Anvil ###
### NatFrance (NFA)-led naval invasion of occupied mainland France, with USA
### assistance once the USA has joined the Entente. Mirrors the Operation
### Homecoming pattern (gated decisions -> flags -> timed commit), but scoped
### to NFA instead of CAN.
###
### Resolved via kaiserreich.wiki: NFA (NatFrance) holds Algeria, Tunisia,
### French West Africa, and Corsica, and never controlled mainland France to
### begin with in this timeline - it's occupied by FRA, the Commune of
### France (syndicalist, confirmed as a distinct tag in this repo's
### localisation and country_tags/00_countries.txt). "Liberation" here means
### an invasion of FRA, not reclaiming lost ground from a wartime occupier.
###
### FIX (from actual game error.log): wotc_anvil_occupier_tag used to be
### defined right here, but scripted triggers can't live inside a decisions
### file - the game tried to parse it as a decision category and threw
### cascading errors through this whole file. It now lives in
### common/scripted_triggers/wotc_triggers.txt.

wotc_operation_anvil = {
	wotc_anvil_request_american_support = {
		icon = GFX_decision_generic_alliance
		available = {
			USA = { is_in_faction_with = ENT }
			NOT = { has_country_flag = wotc_anvil_usa_committed }
		}
		cost = 50

		complete_effect = {
			set_country_flag = wotc_anvil_usa_committed
		}

		ai_will_do = { factor = 1 }
	}

	wotc_anvil_stage_invasion_fleet = {
		icon = GFX_decision_generic_navy
		available = {
			has_country_flag = wotc_anvil_usa_committed
			USA = { is_in_faction_with = ENT } # re-checked in case the USA has since left the faction
			NOT = { has_country_flag = wotc_anvil_fleet_staged }
		}
		days_re_enable = 30 # cooldown if the USA leaves the faction before this completes and it needs retrying
		cost = 100

		complete_effect = {
			set_country_flag = wotc_anvil_fleet_staged
		}

		ai_will_do = { factor = 1 }
	}

	wotc_anvil_launch = {
		icon = GFX_decision_generic_invasion
		available = {
			has_country_flag = wotc_anvil_fleet_staged
			USA = { is_in_faction_with = ENT }
			wotc_anvil_occupier_tag = yes # resolved trigger: country_exists = FRA + has_war_with = FRA
			NOT = { has_country_flag = wotc_anvil_launched }
		}
		cost = 150

		complete_effect = {
			set_country_flag = { flag = wotc_anvil_launched days = 14 }
			add_ideas = wotc_idea_anvil_landing_bonus
		}

		ai_will_do = { factor = 1 }
	}
}
