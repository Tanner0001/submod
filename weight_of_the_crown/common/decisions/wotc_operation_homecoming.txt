### Weight of the Crown - Operation Homecoming (redesigned) ###
###
### Full rework from the original 3-phase version. What changed and why:
###
### - Iceland (state 100, ICE's entire territory - confirmed a single-state
###   minor) no longer gets annexed in one instant decision. Split into an
###   overture (diplomatic buildup, timed) then a separate occupation
###   decision, so there's a real gap between "we're moving in" and
###   "we own it" - addresses it feeling too sudden.
### - Faroe Islands (state 337) and Orkney & Shetland (state 938) are NOT
###   interchangeable "stepping stones" anymore - they're mechanically
###   different, verified from their own history files:
###     - Faroe Islands is owned by DEN (Denmark), a neutral minor, same
###       annex-a-minor shape as Iceland (with its own buildup).
###     - Orkney & Shetland is owned by ENG (Union of Britain) itself - it's
###       enemy home territory, not a minor to annex by decision. It already
###       has `add_claim_by = IMP` in its history file (a scripted claim for
###       the exiled Crown) plus an existing level-8 naval base and a naval
###       supply hub. So this isn't taken by decision at all - it has to be
###       conquered in the actual war, and the decision here just activates
###       once Canada already controls it, formalizing use of infrastructure
###       that's already sitting there.
### - Because of that, "the stepping stones" now means completing both an
###   annexation (Faroe) and an actual battlefield objective (Shetland)
###   before the final landing - more if-conditions, not fewer, per your
###   Operation Anvil feedback that you liked that shape.
### - Phase 3 renamed "Commit to the Landing" is still last and still not
###   auto-detected (no on_action fires on invasion launch), so it stays a
###   deliberate player decision.

wotc_operation_homecoming = {

	### --- Stage 1: Iceland --- ###

	wotc_ice_overture = {
		icon = GFX_decision_generic_agreement
		available = {
			NOT = { 100 = { owner = CAN } }
			NOT = { has_country_flag = wotc_ice_overture_done }
			NOT = { has_country_flag = wotc_ice_overture_pending }
		}
		cost = 30

		complete_effect = {
			set_country_flag = wotc_ice_overture_done
			set_country_flag = { flag = wotc_ice_overture_pending days = 45 } # the buildup window
		}

		ai_will_do = { factor = 1 }
	}

	wotc_ice_occupy = {
		icon = GFX_decision_generic_radar
		available = {
			has_country_flag = wotc_ice_overture_done
			NOT = { has_country_flag = wotc_ice_overture_pending } # the 45-day buildup has elapsed
			NOT = { 100 = { owner = CAN } }
		}
		cost = 50

		complete_effect = {
			set_country_flag = wotc_homecoming_iceland_secured
			add_ideas = wotc_idea_north_sea_intel # permanent naval intel buff, defined alongside the other ideas

			transfer_state = 100 # Iceland - full annexation, see file header
			100 = {
				add_building_construction = {
					type = naval_base
					level = 2 # tops up the existing level-3 base at this province to 5; tune to taste
					instant_build = yes
					province = 12674 # confirmed from history/states/100 - Iceland.txt, already has a level-3 naval base
				}
			}
		}

		ai_will_do = { factor = 1 }
	}

	### --- Stage 2a: Faroe Islands (annex, same shape as Iceland) --- ###

	wotc_faroe_overture = {
		icon = GFX_decision_generic_agreement
		available = {
			has_country_flag = wotc_homecoming_iceland_secured
			NOT = { 337 = { owner = CAN } }
			NOT = { has_country_flag = wotc_faroe_overture_done }
			NOT = { has_country_flag = wotc_faroe_overture_pending }
		}
		cost = 30

		complete_effect = {
			set_country_flag = wotc_faroe_overture_done
			set_country_flag = { flag = wotc_faroe_overture_pending days = 30 }
		}

		ai_will_do = { factor = 1 }
	}

	wotc_faroe_annex = {
		icon = GFX_decision_generic_naval
		available = {
			has_country_flag = wotc_faroe_overture_done
			NOT = { has_country_flag = wotc_faroe_overture_pending }
			NOT = { 337 = { owner = CAN } }
		}
		cost = 40

		complete_effect = {
			set_country_flag = wotc_homecoming_faroe_secured
			transfer_state = 337 # Faroe Islands - owned by DEN per history/states/337 - Faroe Islands.txt
			337 = {
				add_building_construction = {
					type = naval_base
					level = 2 # tops up the existing level-1 base at this province
					instant_build = yes
					province = 13003
				}
			}
		}

		ai_will_do = { factor = 1 }
	}

	### --- Stage 2b: Orkney & Shetland (real invasion objective, not annexed by decision) --- ###

	wotc_shetland_secure_footing = {
		icon = GFX_decision_generic_invasion
		available = {
			has_country_flag = wotc_homecoming_iceland_secured
			938 = { is_controlled_by = CAN } # must actually be conquered in the war first - see file header
			NOT = { has_country_flag = wotc_homecoming_shetland_secured }
		}
		cost = 20 # cheap - the hard part was winning the state, not this paperwork

		complete_effect = {
			set_country_flag = wotc_homecoming_shetland_secured
			add_ideas = wotc_idea_shetland_staging_bonus
			# State 938 already carries a level-8 naval base and a naval
			# supply hub in its own history file - this doesn't build
			# anything new, it just formalizes using what's already there.
		}

		ai_will_do = { factor = 1 }
	}

	### --- Stage 3: Commit to the Landing --- ###

	wotc_phase3_commit_to_the_landing = {
		icon = GFX_decision_generic_invasion
		available = {
			has_country_flag = wotc_homecoming_faroe_secured
			has_country_flag = wotc_homecoming_shetland_secured
			NOT = { has_country_flag = wotc_homecoming_phase_3_active }
		}
		cost = 150

		complete_effect = {
			set_country_flag = { flag = wotc_homecoming_phase_3_active days = 14 }
			add_ideas = wotc_idea_homecoming_landing_bonus_peak # days 1-4, the beachhead
			country_event = { id = wotc_canada.20 days = 4 } # swaps to the days 5-14 sustain idea, then removes it at day 14
		}

		ai_will_do = { factor = 1 }
	}
}
