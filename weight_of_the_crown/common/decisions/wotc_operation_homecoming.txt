### Weight of the Crown - Operation Homecoming ###

wotc_operation_homecoming = {
	icon = GFX_decision_category_generic

	visible = {
		wotc_is_canada = yes
		has_war_with = ENG # confirm actual UoB tag - assumed ENG (Union of Britain) per KR convention
	}

	# Iceland (ICE) is a single-state minor - state 100 is its entire
	# territory (confirmed in history/states/100 - Iceland.txt), so taking it
	# is a full annexation, not a slice of a larger country. Per design
	# decision, that's intended: this models Canada moving in on Iceland the
	# way the Allies did in 1940, just ending in outright control rather
	# than a basing agreement. State 100 already has a level-3/level-1 naval
	# base in its starting history, which we top up rather than build from 0.
	#
	# NOTE: this gates on who currently owns state 100, not on
	# `country_exists = ICE`. Checked Iceland's own focus tree
	# (ICE_Sca_Defense_Union) - it only has Iceland *ask to join* a Nordic
	# faction (ask_to_join_faction), which keeps ICE fully independent and
	# still owning state 100. There's no scripted merger that hands Iceland's
	# territory to Scandinavia. But nothing stops ordinary wartime play from
	# letting someone else (SCA included) conquer Iceland the normal way, so
	# gating on ICE specifically would permanently lock this decision out if
	# that happened. Checking the state's actual owner instead means this
	# still works (takes it from whoever holds it) regardless of how Iceland
	# got there.
	wotc_phase1_reconnaissance = {
		icon = GFX_decision_generic_radar
		available = {
			NOT = { has_country_flag = wotc_homecoming_phase_1_done }
			NOT = { 100 = { owner = CAN } }
		}
		cost = 50

		complete_effect = {
			set_country_flag = wotc_homecoming_phase_1_done
			add_ideas = wotc_idea_north_sea_intel # permanent naval intel buff, defined alongside the other ideas

			transfer_state = 100 # Iceland - see note above on what this means for ICE as a country
			100 = {
				add_building_construction = {
					type = naval_base
					level = 2 # tops up the existing level-3 base at this province to 5; tune to taste
					instant_build = yes
					province = 12674 # confirmed from history/states/100 - Iceland.txt, already has a level-3 naval base
				}
			}
		}

		ai_will_do = { factor = 1 }
	}

	wotc_phase2_stepping_stones = {
		icon = GFX_decision_generic_navy_invasion
		available = {
			has_country_flag = wotc_homecoming_phase_1_done
			NOT = { has_country_flag = wotc_homecoming_phase_2_done }
		}
		cost = 100
		days_remove = 21 # the "timed mission" window itself

		complete_effect = {
			set_country_flag = wotc_homecoming_phase_2_done
			# TODO: state-target the Faroe/Shetland states directly with a
			# temporary supply-hub-range modifier - needs
			# state_target_trigger, this is a country-scope stub.
		}

		ai_will_do = { factor = 1 }
	}

	wotc_phase3_commit_to_the_landing = {
		icon = GFX_decision_generic_invasion
		available = {
			has_country_flag = wotc_homecoming_phase_2_done
			NOT = { has_country_flag = wotc_homecoming_phase_3_active }
		}
		cost = 150

		complete_effect = {
			set_country_flag = { flag = wotc_homecoming_phase_3_active days = 14 }
			add_ideas = wotc_idea_homecoming_landing_bonus
			# Phase 3's bonus is meant to ramp rather than sit flat for the
			# whole window - stagger it with a couple of scheduled effects,
			# e.g. a stronger idea for days 1-4 swapped for a lighter one for
			# the remainder, rather than one flat 14-day modifier.
		}

		ai_will_do = { factor = 1 }
	}
}
