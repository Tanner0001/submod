### Weight of the Crown - Entente High Command ###

wotc_entente_command = {
	wotc_toggle_shared_arsenal = {
		icon = GFX_decision_generic_construction
		available = {
			is_faction_leader = yes
		}

		complete_effect = {
			if = {
				limit = { has_idea = wotc_idea_shared_arsenal_active }
				remove_ideas = wotc_idea_shared_arsenal_active
			}
			else = {
				add_ideas = wotc_idea_shared_arsenal_active
			}
		}

		ai_will_do = { factor = 1 }
	}

	# Recurring drawdown while Shared Arsenal is active. Re-enables itself
	# via days_re_enable so it functions as the "repeating decision" that
	# actually moves equipment, rather than a one-off transfer.
	wotc_shared_arsenal_transfer = {
		icon = GFX_decision_generic_send_supplies

		available = {
			has_idea = wotc_idea_shared_arsenal_active
		}

		targets = { NFA AST } # NatFrance / Australasia - fixed: AUS is Austria, AST is the real Australasia tag (confirmed via localisation)
		target_trigger = {
			FROM = { is_in_faction_with = ENT }
		}

		cost = 50 # PP
		days_re_enable = 30

		complete_effect = {
			# NOTE: this is a faked transfer (stockpile debit/credit), not the
			# real lend-lease system - see the GDD review caveat. Tune the
			# equipment type/amount to whatever's actually excess for Canada.
			add_equipment_to_stockpile = {
				type = infantry_equipment_1
				amount = -500
				producer = CAN
			}
			FROM = {
				add_equipment_to_stockpile = {
					type = infantry_equipment_1
					amount = 500
					producer = CAN
				}
			}
		}

		ai_will_do = { factor = 1 }
	}

	# Rescoped from the original "prioritize a naval zone/garrison a port"
	# idea: no add_ai_strategy type for that exists anywhere in this repo
	# (checked every usage - real types are prepare_for_war, antagonize,
	# support, send_volunteers_desire, send_lend_lease_desire,
	# dont_defend_ally_borders, spare_unit_factor, conquer, declare_war,
	# diplo_action_desire/acceptance - nothing zone- or garrison-targeted).
	# `prepare_for_war` (real, confirmed via this repo's own
	# common/decisions/CAN decisions (Canada).txt) biases an ally's war
	# preparation posture toward a target country - reframed as "direct
	# an ally's war effort against Britain specifically," which is a
	# real lever instead of an invented one.
	wotc_objective_mandate = {
		icon = GFX_decision_generic_navy

		available = {
			is_faction_leader = yes
		}

		targets = { NFA AST }
		target_trigger = {
			FROM = { is_in_faction_with = ENT }
		}

		cost = 75
		days_remove = 60

		complete_effect = {
			FROM = {
				add_ai_strategy = {
					type = prepare_for_war
					id = ENG
					value = 50
				}
			}
		}

		ai_will_do = { factor = 1 }
	}

	# Checked how HOI4 decisions actually target states (there is no
	# `state_target_trigger` - that name doesn't exist anywhere in this repo
	# or the vanilla decision system). The real mechanism is `highlight_states`
	# for cosmetic map highlighting plus a hardcoded state id per decision
	# (see common/decisions/SER decisions (Serbia).txt's SER_IMRO_reb_lull_phase),
	# not a free-form "player picks any state" feature - so this can't be
	# one generic decision the way it's written. It needs to become several
	# concrete decisions, one per real target state (the GDD's own example
	# was Algiers/Malta), each with a hardcoded state id. That's a design
	# choice (which states, matching which ally) I haven't made rather than
	# a syntax gap, so it's left as a country-wide bonus for now instead of
	# a fake per-state effect.
	wotc_joint_invasion_prep = {
		icon = GFX_decision_generic_invasion

		available = {
			has_idea = wotc_idea_entente_cohesion_high
		}

		cost = 100
		days_remove = 14

		complete_effect = {
			add_ideas = wotc_idea_shetland_staging_bonus # reuses the real, verified supply_consumption_factor idea as a stand-in country-wide logistics bonus until real per-state targets are chosen
		}

		ai_will_do = { factor = 1 }
	}
}
