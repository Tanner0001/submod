### e.g. annexation_decisions).
###
### FIX 2 (from in-game report): all five categories were showing at once
### regardless of country/state, even though each has gating conditions
### below. `available` does NOT hide a decision category tab - confirmed by
### comparing against this repo's own `lar_local_recruitment` category
### (common/decisions/categories/00 Generic decision_cats.txt), which uses
### `visible` for exactly this purpose. Switched every category below from
### `available` to `visible`.

wotc_liberation_of_allies = {
	icon = GFX_decision_category_generic
	priority = 100

	# Visible once an ENT ally has fallen (wotc_ally_needs_sanctuary) and
	# stays visible through the rest of the chain (wotc_hosting_ally_exile) -
	# gating on just the first flag would hide the category the moment
	# Offer Sanctuary clears it, before Rally/Declare could ever be reached.
	visible = {
		wotc_is_canada = yes
		OR = {
			has_country_flag = wotc_ally_needs_sanctuary
			has_country_flag = wotc_hosting_ally_exile
		}
	}
}
