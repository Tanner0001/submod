### Weight of the Crown - on_actions ###
### These blocks merge with the base game's/Kaiserreich's on_actions of the
### same name - do not duplicate on_startup/on_state_control_changed logic
### that already exists elsewhere in the load order.

on_actions = {
	on_startup = {
		effect = {
			if = {
				limit = { wotc_is_canada = yes }
				wotc_init_variables = yes
				wotc_reapply_exile_tier = yes
				wotc_reapply_tension_tier = yes
				country_event = { id = wotc_canada.1 days = 5 }
			}
		}
	}

	# Fires in the scope of the country that just gained control of a state.
	# FROM = previous controller, FROM.FROM = the state itself. Tension is
	# tracked on Canada regardless of which country this fires for, so we
	# switch scope into CAN explicitly rather than gating on wotc_is_canada.
	on_state_control_changed = {
		effect = {
			if = {
				limit = {
					FROM.FROM = { wotc_is_49th_parallel_state = yes }
				}
				FROM.FROM = { save_temp_scope_as = wotc_changed_state }
				CAN = {
					if = {
						limit = {
							temp_scope:wotc_changed_state = { wotc_state_controller_is_hostile_to_canada = yes }
						}
						add_to_variable = { wotc_border_tension = 8 }
					}
					else = {
						subtract_from_variable = { wotc_border_tension = 4 }
					}
					wotc_reapply_tension_tier = yes
				}
			}
		}
	}
}
