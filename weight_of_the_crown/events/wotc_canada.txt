add_namespace = wotc_canada

# Hidden pulse event: reschedules itself every 5 days on Canada to decay
# border tension and occasionally fire a flavor incident. This is the
# standard Paradox-modding idiom for "run this periodically" - there is no
# generic on_action for that, so a self-rescheduling hidden event is the
# correct tool, not a shortcut.
country_event = {
	id = wotc_canada.1
	hidden = yes
	is_triggered_only = yes

	immediate = {
		wotc_decay_border_tension = yes
		wotc_reapply_tension_tier = yes

		# FIX: `random_events` isn't a real effect - replaced with
		# `random_list`, confirmed via common/national_focus/ALB focus
		# (Albania).txt, where each weight maps to an effect block, not a
		# bare event id.
		if = {
			limit = { wotc_tier_covert_proxy = yes }
			random_list = {
				85 = { } # nothing happens
				15 = { country_event = wotc_canada.10 } # border incident
			}
		}
		else_if = {
			limit = { wotc_tier_intervention = yes }
			random_list = {
				60 = { }
				30 = { country_event = wotc_canada.10 } # border incident
				10 = { country_event = wotc_canada.11 } # draft riot flavor, only meaningful at Exile Dominance
			}
		}

		country_event = { id = wotc_canada.1 days = 5 }
	}
}

# Border incident flavor event - fires more often at higher tension tiers.
country_event = {
	id = wotc_canada.10
	title = wotc_canada.10.t
	desc = wotc_canada.10.d
	picture = GFX_report_event_generic_war

	is_triggered_only = yes

	option = {
		name = wotc_canada.10.a
		add_to_variable = { wotc_border_tension = 5 }
		add_stability = -0.02
	}
	option = {
		name = wotc_canada.10.b
		add_political_power = -50
		add_to_variable = { wotc_border_tension = -3 }
	}
}

# Draft riot flavor - only reachable while in the Exile Dominance tier, per
# the GDD's "potential for draft riots" consequence.
country_event = {
	id = wotc_canada.11
	title = wotc_canada.11.t
	desc = wotc_canada.11.d
	picture = GFX_report_event_civil_unrest

	is_triggered_only = yes

	trigger = {
		wotc_tier_exile_dominance = yes
	}

	option = {
		name = wotc_canada.11.a
		add_stability = -0.05
		add_to_variable = { wotc_exile_influence = -5 }
		wotc_reapply_exile_tier = yes
	}
	option = {
		name = wotc_canada.11.b
		# Hooks into KR's own existing Quebec unrest system rather than
		# inventing a new one - confirmed real via events/CAN events
		# (Canada).txt's CAN_war_measures_act chain, which applies
		# CAN_quebecois_resistance_modifier the same way. QUE is a real
		# country tag (common/country_tags/00_countries.txt).
		QUE = {
			every_core_state = {
				limit = {
					is_owned_by = CAN
					impassable = no
				}
				custom_override_tooltip = {
					tooltip = {
						localization_key = tooltip_add_dynmod
						DYNMOD = CAN_quebecois_resistance_modifier
					}
					add_dynamic_modifier = { modifier = CAN_quebecois_resistance_modifier }
				}
				add_to_variable = { CAN_qr_state_resources_factor = -0.2 }
				add_to_variable = { CAN_qr_local_manpower = -0.2 }
				add_to_variable = { CAN_qr_local_factory_sabotage = 0.1 }
			}
		}
	}
}

# Operation Homecoming Phase 3 ramp, part 1: swaps the peak beachhead bonus
# for the lighter sustain bonus 4 days after the landing decision fires.
# Scheduled from wotc_operation_homecoming.txt's wotc_phase3_commit_to_the_landing.
country_event = {
	id = wotc_canada.20
	hidden = yes
	is_triggered_only = yes

	immediate = {
		if = {
			limit = { has_idea = wotc_idea_homecoming_landing_bonus_peak }
			remove_ideas = wotc_idea_homecoming_landing_bonus_peak
			add_ideas = wotc_idea_homecoming_landing_bonus
		}
		country_event = { id = wotc_canada.21 days = 10 } # remaining 10 of the 14-day window
	}
}

# Part 2: removes the sustain bonus once the full 14-day window has elapsed.
country_event = {

	is_triggered_only = yes

	immediate = {
		if = {
			limit = { has_idea = wotc_idea_homecoming_landing_bonus }
			remove_ideas = wotc_idea_homecoming_landing_bonus
		}
	}
}

# The Fate of [Ally]'s Homeland - fires once Canada personally defeats the
# conqueror tracked by wotc_declare_liberation (common/on_actions/
# wotc_on_actions.txt's on_capitulation extension). Mirrors Kaiserreich's
# own "The Fate of America" (american_civil_war_events.156 in
# events/01 American Civil War events.txt) - a real Canada-scoped event
# doing the same give-the-territory-to-X choice, not a decision.
country_event = {
	id = wotc_canada.40
	title = wotc_canada.40.t
	desc = wotc_canada.40.d
	picture = GFX_report_event_generic_conference

	is_triggered_only = yes

	immediate = {
		clr_country_flag = wotc_hosting_ally_exile
		clr_country_flag = wotc_legions_rallied
		clr_country_flag = wotc_liberation_war_active
		if = {
			limit = { has_idea = wotc_idea_free_legions_support }
			remove_ideas = wotc_idea_free_legions_support
		}
		if = {
			limit = { has_idea = wotc_idea_liberation_offensive_bonus }
			remove_ideas = wotc_idea_liberation_offensive_bonus
		}
	}

	option = {
		name = wotc_canada.40.a # Restore Independence
		release = event_target:wotc_fallen_ally
		add_stability = 0.03
	}
	option = {
		name = wotc_canada.40.b # Establish a Protectorate
		release_puppet = event_target:wotc_fallen_ally
		add_political_power = -75
		add_stability = -0.02
	}
	option = {
		name = wotc_canada.40.c # Absorb the Conquerors
		puppet = { target = event_target:wotc_liberation_target }
		add_political_power = -150
		add_stability = -0.05
		every_other_country = {
			limit = { is_in_faction_with = ENT }
			add_opinion_modifier = { target = CAN modifier = KR_disapproves_of_govt }
		}
	}
}
